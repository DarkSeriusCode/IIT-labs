# Упражнение 3.5

Для того чтобы добавить нового пользователя и сделать выполнение комманд от его имени, изменим 
Dockerfile для каждого образа

Для frontend:
```Dockerfile
FROM node:16.0

EXPOSE 5000

WORKDIR /usr/src/app

ENV REACT_APP_BACKEND_URL=http://localhost:8080

COPY . .

RUN npm install
RUN npm install -g serve

RUN npm run build

RUN useradd -m appuser

USER appuser

CMD ["serve", "-s", "-l", "5000", "build"]
```

Для backend:
```
FROM golang:1.16.15

EXPOSE 8080

ENV PORT=8080
ENV REQUEST_ORIGIN=http://localhost:5000

WORKDIR /usr/src/app

COPY . .

RUN go build

RUN adduser appuser
USER appuser

CMD ["./server"]
```
