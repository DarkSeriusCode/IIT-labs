# Упражнение 2.10

Наше приложение использует следующий `docker-compose.yml` файл:
```yaml
services:
  frontend:
    image: rustykrabik/frontend
    build:
      context: example-frontend
      args:
        REACT_APP_BACKEND_URL: http://localhost/api

  backend:
    image: rustykrabik/backend
    build: example-backend
    environment:
      - REQUEST_ORIGIN=http://localhost/
      - REDIS_HOST=redis
      - POSTGRES_HOST=db
      - POSTGRES_USER=God
      - POSTGRES_PASSWORD=Jesus
    depends_on:
      - redis
      - db

  redis:
    image: redis:8

  db:
    image: postgres:13.2-alpine
    restart: unless-stopped
    environment:
      - POSTGRES_USER=God
      - POSTGRES_PASSWORD=Jesus
    volumes:
      - database:/var/lib/postgresql/data

  proxy:
    image: nginx
    restart: unless-stopped
    ports:
      - 80:80
    depends_on:
      - frontend
      - backend
    volumes:
      - ./conf/nginx.conf:/etc/nginx/nginx.conf


volumes:
  database:
```

Давайте найдём открытые порты и избавимся от них. Запустим сканер портов:
```
docker run -it --rm --network host networkstatic/nmap localhost
```

